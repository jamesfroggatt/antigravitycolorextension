import { createServer } from '@antigravity/core';
import { verifySignature } from './security/hmac';
// Code Editor background ("editor.background") updates live via Surface Sliders
export async function handleWebhook(req: Request) {
const sig = req.headers.get('x-antigravity-sig');
if (!sig || !verifySignature(req.body, sig)) {
return new Response('Unauthorized', { status: 401 });
}
return Response.json({ received: true, timestamp: Date.now() });
}
